128 void spill(int c) noexcept {
129 std::vector<void *> &mine = spare_[c];
130 mine.push_back(free_[c]);
131 free_[c] = nullptr;
132 count_[c] = 0;
133 if (mine.size() > kLocalBatches) {
134 // Pass the older half: the newest batches are the likeliest in cache
135 const size_t keep = kLocalBatches / 2;
136 const size_t give = mine.size() - keep;
137 Pool &p = pool();
138 std::lock_guard<std::mutex> lock{p.mutex};
139 p.batches[c].insert(p.batches[c].end(), mine.begin(),
140 mine.begin() + give);
141 p.available[c].store(static_cast<int32_t>(p.batches[c].size()),
142 std::memory_order_relaxed);
143 mine.erase(mine.begin(), mine.begin() + give);
144 }
145 }

149 bool refill(int c) {
150 std::vector<void *> &mine = spare_[c];
151 if (mine.empty()) {
152 Pool &p = pool();
153 // A stale read only means carving one block more, or taking the mutex
154 // and finding the pool empty after all; both are safe
155 if (p.available[c].load(std::memory_order_relaxed) == 0)
156 return false;
157 std::lock_guard<std::mutex> lock{p.mutex};
158 if (p.batches[c].empty())
159 return false;
160 mine.push_back(p.batches[c].back());
161 p.batches[c].pop_back();
162 p.available[c].store(static_cast<int32_t>(p.batches[c].size()),
163 std::memory_order_relaxed);
164 }
165 free_[c] = mine.back();
166 mine.pop_back();
167 count_[c] = kBatch;
168 return true;
169 }
