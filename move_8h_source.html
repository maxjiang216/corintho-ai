43constexpr std::array<MoveInfo, kNumMoves> makeMoveTable() {
44 std::array<MoveInfo, kNumMoves> table{};
45 for (int32_t id = 0; id < kNumMoves; ++id) {
46 if (id >= 48) { // Place
47 table[id] =
48 MoveInfo{-1, static_cast<int8_t>(((id % 16) / 4) * 4 + id % 4),
49 static_cast<int8_t>((id - 48) / 16), true};
50 } else if (id < 12) { // Right
51 const int32_t r = id / 3, c = id % 3;
52 table[id] = MoveInfo{static_cast<int8_t>(r * 4 + c),
53 static_cast<int8_t>(r * 4 + c + 1), -1, false};
54 } else if (id < 24) { // Down
55 const int32_t r = (id - 12) / 4, c = id % 4;
56 table[id] = MoveInfo{static_cast<int8_t>(r * 4 + c),
57 static_cast<int8_t>((r + 1) * 4 + c), -1, false};
58 } else if (id < 36) { // Left
59 const int32_t r = (id - 24) / 3, c = id % 3 + 1;
60 table[id] = MoveInfo{static_cast<int8_t>(r * 4 + c),
61 static_cast<int8_t>(r * 4 + c - 1), -1, false};
62 } else { // Up
63 const int32_t r = (id - 36) / 4 + 1, c = id % 4;
64 table[id] = MoveInfo{static_cast<int8_t>(r * 4 + c),
65 static_cast<int8_t>((r - 1) * 4 + c), -1, false};
66 }
67 }
68 return table;
69}

101constexpr std::array<std::array<MoveMask, 3>, kNumLineShapes>
102makeLineBreakTable() {
103 std::array<std::array<MoveMask, 3>, kNumLineShapes> table{};
104 for (int32_t s = 0; s < kNumLineShapes; ++s) {
105 const LineShape &shape = kLineShapes[s];
106 for (int32_t type = 0; type < 3; ++type) {
107 MoveMask mask{};
108 for (int32_t id = 0; id < kNumMoves; ++id) {
109 const MoveInfo &move = kMoveTable[id];
110 bool from_in_line = false;
111 bool to_in_line = false;
112 for (int32_t k = 0; k < shape.count; ++k) {
113 if (move.from == shape.cells[k])
114 from_in_line = true;
115 if (move.to == shape.cells[k])
116 to_in_line = true;
117 }
118 if (move.is_place) {
119 if (to_in_line && move.piece != type)
120 mask.set(id); // retypes a line space
121 if (shape.extend >= 0 && move.to == shape.extend &&
122 move.piece == type)
123 mask.set(id); // completes a four
124 } else if (from_in_line || to_in_line) {
125 mask.set(id); // empties a line space, or lands on one
126 }
127 }
128 table[s][type] = mask;
129 }
130 }
131 return table;
132}
