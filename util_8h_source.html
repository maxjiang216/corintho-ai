126 return i < 64 ? ((lo >> i) & 1U) != 0 : ((hi >> (i - 64)) & 1U) != 0;
127 }
128 constexpr void set(int32_t i) noexcept {
129 if (i < 64)
130 lo |= 1ULL << i;
131 else
132 hi |= 1ULL << (i - 64);

135 if (i < 64)
136 lo &= ~(1ULL << i);
137 else
138 hi &= ~(1ULL << (i - 64));
139 }
140 constexpr bool any() const noexcept { return (lo | hi) != 0; }
141 int32_t count() const noexcept {
142 return __builtin_popcountll(lo) + __builtin_popcountll(hi);
143 }

145 lo &= o.lo;

150 lo |= o.lo;

262makeRunToShape() {
263 std::array<std::array<std::array<int8_t, 2>, kBoardSize>, 4> table{};
264 for (int32_t d = 0; d < 4; ++d)
265 for (int32_t c = 0; c < kBoardSize; ++c)
266 for (int32_t n = 0; n < 2; ++n)
267 table[d][c][n] = -1;
268 for (int32_t s = 0; s < kNumLineShapes; ++s) {
269 const LineShape &shape = kLineShapes[s];
270 int32_t lo = shape.cells[0];
271 for (int32_t k = 1; k < shape.count; ++k)
272 if (shape.cells[k] < lo)
273 lo = shape.cells[k];
274 int32_t second = 64;
275 for (int32_t k = 0; k < shape.count; ++k)
276 if (shape.cells[k] > lo && shape.cells[k] < second)
277 second = shape.cells[k];
278 const int32_t stride = second - lo;
279 for (int32_t d = 0; d < 4; ++d)
280 if (kLineStride[d] == stride)
281 table[d][lo][shape.count - 3] = static_cast<int8_t>(s);
282 }
283 return table;
284}
